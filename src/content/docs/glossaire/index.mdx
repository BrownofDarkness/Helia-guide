import { glossaire } from '../../../data/glossaire';
import GlossaryFilterScript from '../../../components/GlossaryFilterScript.astro';

export const AXE_SLUGS = {
  '1': 'fondations',
  '2': 'web',
  '3': 'analyse-conception',
  '4': 'outils',
  '5': 'frontend-html-css',
  '6': 'javascript-typescript',
  '7': 'frameworks-frontend',
  '8': 'backend',
  '9': 'bases-de-donnees',
  '10': 'baas',
  '11': 'qualite-tests',
  '12': 'securite',
  '13': 'performance-a11y',
  '14': 'devops',
  '15': 'methodes-soft-skills',
  '16': 'specialisations',
  '17': 'carriere',
};

export const sortedEntries = Object.entries(glossaire).sort(([a], [b]) =>
  a.localeCompare(b, 'fr', { sensitivity: 'base' })
);

export const letterFirstKey = sortedEntries.reduce((acc, [key]) => {
  const letter = key[0].toLowerCase();
  if (!acc[letter]) acc[letter] = key;
  return acc;
}, {});

export const ALPHABET = 'abcdefghijklmnopqrstuvwxyz'.split('');

export const presentAxes = [...new Set(
  Object.values(glossaire).map((e) => e.axe?.split('.')[0]).filter(Boolean)
)].sort((a, b) => Number(a) - Number(b));

> 🎯 **Comment l'utiliser** : dans le guide, certains mots sont <span class="term">soulignés en pointillés</span>. Survole-les pour la définition rapide ; clique pour atterrir ici sur l'entrée détaillée. Sur cette page, tu peux **filtrer** par axe ou **rechercher** instantanément.

<div class="glossary-controls">
  <label class="glossary-search-label">
    <span class="glossary-search-icon" aria-hidden="true">🔍</span>
    <input
      type="search"
      id="glossary-search"
      placeholder="Rechercher un terme, une définition..."
      aria-label="Rechercher dans le glossaire"
      autocomplete="off"
    />
    <span class="glossary-count" id="glossary-count" aria-live="polite">{sortedEntries.length} termes</span>
  </label>

  <div class="glossary-filters" role="toolbar" aria-label="Filtrer par axe">
    <button type="button" class="axe-pill active" data-axe="all">Tous</button>
    {presentAxes.map((axe) => (
      <button type="button" class="axe-pill" data-axe={axe}>Axe {axe}</button>
    ))}
  </div>

  <nav class="alphabet-nav" aria-label="Aller à une lettre">
    {ALPHABET.map((l) => (
      letterFirstKey[l]
        ? <a href={`#${letterFirstKey[l]}`} data-letter={l}>{l.toUpperCase()}</a>
        : <span class="disabled" aria-disabled="true">{l.toUpperCase()}</span>
    ))}
  </nav>
</div>

<section class="glossary-list">
{sortedEntries.map(([key, entry]) => {
  const axeNum = entry.axe?.split('.')[0];
  return (
    <div
      id={key}
      class="glossary-entry"
      data-axe={axeNum ?? ''}
      data-letter={key[0].toLowerCase()}
      data-key={key.toLowerCase()}
      data-short={entry.short.toLowerCase()}
    >
      <h3>{key}</h3>
      <p>{entry.short}</p>
      {entry.exemple && <p class="glossary-example"><strong>Exemple :</strong> {entry.exemple}</p>}
      {entry.axe && (
        <p class="glossary-axe">
          Développé dans <a href={`/${axeNum}-${AXE_SLUGS[axeNum]}/`}>axe {entry.axe}</a>
        </p>
      )}
    </div>
  );
})}
</section>

<p class="glossary-empty" id="glossary-empty" hidden>
  Aucun terme ne correspond. Réinitialise les filtres ou modifie ta recherche.
</p>
